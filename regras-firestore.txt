rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function admin() {
      return request.auth != null && request.auth.uid == "7Vfs8c3XevgZLcZSOwXBmUg5Zue2";
    }
    match /mapas/{id} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null
        && exists(/databases/$(database)/documents/objetivos/$(id))
        && get(/databases/$(database)/documents/objetivos/$(id)).data.modo in ['distancia', 'direcao']
        && request.resource.data.keys().hasOnly(['desenhos'])
        && request.resource.data.desenhos is list
        && request.resource.data.desenhos.size() <= 100;
      allow delete: if admin();
    }
    match /objetivos/{id} {
      allow read: if request.auth != null;
      allow create, delete: if admin();
      allow update: if admin() || (
        request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['encontrado'])
        && request.resource.data.encontrado == true
      );
    }
  }
}
